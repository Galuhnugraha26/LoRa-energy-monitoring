//Initialize LCD display 
void startLCD(){ 
LiquidCrystal_I2C lcd(0x27, 16, 2); 
lcd.begin(); 
lcd.backlight(); 
lcd.setCursor(0, 0); 
lcd.print("Energy Meter"); 
lcd.setCursor(0, 1); 
lcd.print("With LoRa"); 
delay(2000); 
lcd.clear(); 
} 
//Initialize LoRa module 
void startLoRA(){ 
//SPI LoRa pins 
SPI.begin(SCK, MISO, MOSI, SS); 
//setup LoRa transceiver module 
LoRa.setPins(SS, RST, DIO0); 
while (!LoRa.begin(BAND) && counter < 10) { 
Serial.print("."); 
counter++; 
delay(500); 
} 
if (counter == 10) { 
// Increment readingID on every new reading 
readingID++; 
Serial.println("Starting LoRa failed!");  
} 
Serial.println("LoRa Initialization OK!"); 
lcd.setCursor(0, 0); 
lcd.clear(); 
lcd.print("LoRa OK!"); 
delay(2000); 
}

void getTOTAL(){ 
// electricity cost estimation calculation (PLN adjustment 
tariff Jan-Mar 2023) 
float cost = pzem1.energy() * hargaPerKWH; // get the rupiah 
value from energy1 
float cost2 = pzem2.energy() * hargaPerKWH ; // get the rupiah 
value from energy2 
// Total sum of estimated costs from energymeter1 and 
energymeter2 
float total_cost = (cost + cost2) * 100; 
Serial.print("Total Cost: ");   
Serial.print(total_cost); 
Serial.println("Rupiah"); 
total_costStr = String (total_cost = (cost + cost2) * 100); 
} 
void sendReadings() { 
LoRaMessage = String (readingID) + "," + voltageStr + "," +  
currentStr + "," + powerStr + "," + energyStr + ","  
+ voltage2Str + "," +  current2Str + "," + power2Str + "," + 
energy2Str + ","  
+ total_powerStr +  "," + total_costStr; 
//Send LoRa packet to receiver 
LoRa.beginPacket(); 
LoRa.print(LoRaMessage); 
LoRa.endPacket(); 
readingID++; 
} 
void setup() { 
//initialize Serial Monitor 
Serial.begin(115200); 
pzem1.resetEnergy(); 
pzem2.resetEnergy(); 
startLCD(); 


  startPZEM(); 
  startLoRA(); 
} 
void loop() { 
  getReadings(); 
  getTOTAL(); 
  sendReadings(); 
  delay(10000); 
}
