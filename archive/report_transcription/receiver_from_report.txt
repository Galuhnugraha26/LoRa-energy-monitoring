//Initialize LoRa module 
void startLoRA(){ 
int counter; 
//SPI LoRa pins

//setup LoRa transceiver module 
LoRa.setPins(SS, RST, DIO0); 
while (!LoRa.begin(BAND) && counter < 10) { // initiation of 
lora module connected with frequency 923 
Serial.print("."); 
counter++; 
delay(500); 
} 
if (counter == 10) { 
// Increment readingID on every new reading 
Serial.println("Starting LoRa failed!");  
} 
Serial.println("LoRa Initialization OK!"); 
lcd.setCursor(0,0); 
lcd.print("LoRa OK!"); 
delay(2000); 
lcd.clear(); 
}
